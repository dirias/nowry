import { useEffect, useState } from 'react'
import { useQuery } from '@tanstack/react-query'
import { cardsService } from '../api/services'
import { queryClient } from '../api/queryClient'
import { useAuth } from '../context/AuthContext'
 
/** The sentinel the deck-id slot carries for "today, across every deck". */
export const DAILY_REVIEW = 'daily-review'
 
/**
 * How long an opening session waits for the server before it settles for the
 * cards on the device (STUDY-012). Long enough for a normal answer; short
 * enough that a dead signal still studies from the cache.
 */
export const OPEN_WAIT_MS = 3000
 
/**
 * The cards a session is about to ask (MOB-026, MOB-042).
 *
 * A `useQuery` rather than a bare service call, and that is the whole point:
 * the mobile client persists its query cache to disk, so a queue fetched with
 * signal is a queue that opens without one. Fetching directly — which is what
 * this screen did — put the one part of the app that most needs to work offline
 * outside the only thing that makes offline possible.
 *
 * `deckId` is either a deck's id or `daily-review`, and the daily review is a
 * queue the SERVER owns: it stamps the day's selection, so the same cards come
 * back until they are graded rather than being redrawn on each open. `tags` and
 * `group` narrow that pool server-side.
 *
 * The key carries every narrowing, so a tag's session and the whole day's are
 * separate entries and neither serves the other's cards from cache.
 */
/**
 * @param {object} options
 * @param {'due' | 'all'} [options.scope] - `all` is every card of the deck, the web's browse
 *   mode and the phone's All scope (ADR-042); ignored for the daily review, which has no All.
 * @param {boolean} [options.enabled] - a second scope can be mounted without being fetched.
 */
export function useSessionCards({ deckId, tags = [], group, limit, attempt = 0, scope = 'due', enabled = true } = {}) {
  const { user } = useAuth()
  const userId = user?.id ?? null
  const isDaily = deckId === DAILY_REVIEW
 
  const everyCard = scope === 'all' && !isDaily
 
  const { data, isLoading, error, isFetching } = useQuery({
    queryKey: ['cards', userId, 'session', deckId, { tags, group, limit, attempt, scope: everyCard ? 'all' : 'due' }],
    queryFn: () =>
      isDaily
        ? cardsService.getDailyReviewCards({ limit, tags, group })
        : everyCard
          ? cardsService.getAllCards(deckId)
          : // A deck's tags narrow its pool before the budget (TAG-002, ADR-044),
            // the way the daily review's have since STUDY-002.
            cardsService.getDueCards(deckId, { tags }),
    enabled: enabled && !!userId && !!deckId,
    /*
     * A new narrowing of the SAME deck keeps the cards on screen until the
     * new pool arrives (MOB-118). Choosing a tag changes the key, and a bare
     * `null` in between sent the session to its loading branch, which
     * unmounted the queue sheet the tag was chosen in and reset it. Another
     * deck never borrows these cards: the key's fourth slot is the deck.
     */
    placeholderData: (previous, previousQuery) => (previousQuery?.queryKey?.[3] === deckId ? previous : undefined),
    /*
     * A session's queue is read once and then worked through. Refetching it
     * mid-session would change the cards under the user's thumb, so it stays
     * fresh for as long as a session plausibly lasts.
     */
    staleTime: 15 * 60 * 1000,
    /*
     * But every OPENING asks the server (STUDY-012, ADR-057 §3). Grading does
     * not touch this entry, so a deck reopened inside the window was served the
     * cards just graded — a finished session came back whole, and grading it
     * again sent a second review for each card.
     */
    refetchOnMount: 'always'
  })
 
  /*
   * Cards from the device are held back until that opening fetch answers, so
   * the session never positions itself in a queue about to be replaced: the
   * phone resumed by card in the cached list, then the fresh one arrived
   * shorter and the same number pointed past unseen cards. A fetch that does
   * not answer within OPEN_WAIT_MS gives the cache back — offline study stays.
   */
  const [opened, setOpened] = useState(false)
  useEffect(() => {
    if (opened) return undefined
    if (!isFetching) {
      setOpened(true)
      return undefined
    }
    const timer = setTimeout(() => setOpened(true), OPEN_WAIT_MS)
    return () => clearTimeout(timer)
  }, [isFetching, opened])
  const opening = !opened && isFetching && Boolean(data)
 
  // Every session answer is wrapped now (the cards, the pool's tags, and for
  // the daily review what the day said: the resting count and the landing);
  // a plain list is still read, for an answer shaped before TAG-001.
  const cards = opening ? null : Array.isArray(data) ? data : (data?.items ?? null)
 
  return {
    cards,
    resting: Array.isArray(data) ? 0 : (data?.resting ?? 0),
    spread: Array.isArray(data) ? null : (data?.spread ?? null),
    /** The pool's tags with something waiting (TAG-002); read through `poolTagRows`. */
    poolTags: Array.isArray(data) ? [] : (data?.tags ?? []),
    loading: isLoading || opening,
    /** True only when there is nothing to show: cached cards beat an error. */
    error: data ? null : (error ?? null),
    /** A refetch is happening over cards already on screen. */
    refreshing: isFetching && Boolean(data) && !opening,
    reload: () => queryClient.invalidateQueries({ queryKey: ['cards', userId, 'session'] })
  }
}
 
export default useSessionCards
